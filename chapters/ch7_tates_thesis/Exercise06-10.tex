\begin{exercise}[6]
  \label{ex:ch7-6}
  We may now use the results of previous problems to construct the standard
  character $\psi_K \colon \AAA_K \to S^1$ for $K$ any global field of
  positive characteristic.
  \begin{enumerate}[label=(\alph*)]
    \item Suppose that $K_0 = \FF_p(t)$. Define $\psi_{K_0}$ by
      \[
        \psi_{K_0}(x) = \prod_v \psi_v(x_v)
      \]
      where $\psi_\infty$ is given by \cref{ex:ch7-3} and $\psi_v$
      ($v$ finite) is given by part~(c) of the previous exercise. Show that
      $\psi_{K_0}$ is a nontrivial character of $\AAA_{K_0}$ that, moreover,
      is trivial on $K_0$.
    \item Now let $K$ be any finite separable extension of $K_0$. Use the
      trace map from $K$ to $K_0$ (or, equivalently, the corresponding adelic
      trace map) to define the standard character $\psi_K$ on $K$. Show that
      this character is nontrivial on $\AAA_K$, but trivial on $K$.
  \end{enumerate}
\end{exercise}

\begin{solution}
  (a) Let $U = K_\infty \times \prod_{v} \gotho_v$. Note that
  $\psi_{K_0}(x) | _U =\psi_\infty(x)$, and $U$ is an open subgroup of
  $\AAA_{K_0}$. And for element $x = (1/t, 0, 0, \ldots, 0) \in \AAA_{K_0}$,
  $\psi_{K_0} (x) = \psi_\infty (1 / t) = e ^ {2 \pi i/ p}$.
  Then $\psi_{K_0}$ is nontrivial.

  Claim: for every $x \in K_0$, $x$ can be written as
  \[x = f + \sum_{i = 0}^{r} \frac{z_i}{\pi_i ^ {n_i}},\]
  where $f \in \FF_p[t]$ and $\deg z_i < n_i d_i$, $d_i = \deg \pi_i$.

  \textit{Proof of claim:} let $x = a / b$, where
  $b = \prod_{i = 1}^{r} \pi_i ^ {n_i}$, where $\pi_i$ are different monic
  irreducible polynomial. Let $b_i = b / \pi_i^{n_i}$ and
  these are coprime. Since $\FF_p[t]$ is principal ideal domain, then
  there exist ${u_i}_{i = 1}^r$ such that $\sum_{i=1}^{r} u_i b_i = 1$.
  Hence $x = \sum_{i =1}^{r} a u_i / \pi_i^{n_i}$. By division algorithm,
  we have that $a u_i = q_i \pi_i ^ {n_i} + z_i$,
  where $\deg z_i < n_i \deg \pi_i$.

  Since $\psi_{K_0}$ is a homomorphism, then we only need to check that
  for $f \in \FF_p[t]$ and $y = z/ \pi ^ n$ satisfy $\psi_{K_0} (f) = 1$ and
  $\psi_{K_0} (y) = 1$.

  \begin{itemize}
    \item For $f \in \FF_p[t]$, $f \in \gotho_v$ for all finite place $v$.
    It follows that $\psi_v(f) = 1$. And at the infinite place $\infty$,
    $\psi_\infty (f) = 1$, since $a_{-1} = 0$.
    \item For $y = z / \pi^ n$ with $\deg z < n d$ and let
    $y \equiv c_1 t ^ {-1} \pmod{t ^ {-2}}$. For $v = v_\pi$,
    $\psi_v (y) = e^{2\pi i c_1 / p}$. And for any other finite place
    $u \neq v$, $\psi_u(y) = 1$. For infinite place $\infty$,
    $\psi_\infty (y) = e^{-2 \pi i c_1 / p}$.
  \end{itemize}

  In conclusion, $\forall x \in K_0$, $\psi_{K_0} (x) = 1$, i.e.
  $\psi_{K_0} | _{K_0} = 1$.

  (b) Recall that adelic tract is defined as follows:
  \[\tr (x)_v = \sum_{w \mid v} \tr_{K_w / K_{0, v}} (x_w). \]

  We define $\psi_K = \psi_{K_0} \circ \tr$ or equivalently
  $\psi_K(x) = \prod_{w} \psi_w (x_w)$, where
  $\psi_w = \psi_v \circ \tr_{K_w/ K_{0, v}}$. Due to separability
  of $K/ K_0$, then $\tr_{K/ K_0}$ is nontrivial and
  hence $\psi_K$ is non-trivial.

  Claim: for any $a \in K$ and every place $v$, then
  $\tr_{K/ K_0} (a) = \sum_{w \mid v} \tr_{K_w/ K_{0, v}} (a)$
  This means that adelic trace restricts to $K$ is global trace,
  that is $\tr |_K = \tr_{K/K_0}$.

  \textit{Proof of claim:} since $K/ K_0$ is separable,
  $K \otimes_{K_0} K_{0, v} \cong \prod_{w \mid v} K_w$.
  The matrix of multiplication by $a$ map in $K$ related to
  $K_0$-basis $\{e_i\}$ is the same as in $K \otimes K_{0, v}$
  as $K_{0, v}$-basis $\{e_i \otimes 1\}$.
  Then

  \[\tr_{K/K_0} (a) = \tr_{(K \otimes K_{0, v})/ K_{0, v}} (a \otimes 1) =
    \sum_{w} \tr_{K_w/ K_{0, v}} (a).  \]

  Then $\psi_K(a) = \psi_{K_0} (\tr_{K/ K_0} a)$. It follows that
  $\tr_{K/K_0} (a) \in K_0$. By $(a)$, $\psi_K (a) = 1$.

\end{solution}

\begin{exercise}[8]
  \label{ex:ch7-8}
  In this exercise we analyze the dependence of the epsilon factor on the
  additive character $\psi$ and the Haar measure $\dd x$. Let $F$ be a local
  field and consider $\varepsilon(\chi, \psi, \dd x)$ for any
  $\chi \in \Hom_{\cont}(F^\times, \CC^\times)$.
  \begin{enumerate}[label=(\alph*)]
    \item For every positive real number $t$, show that
      \[
        \varepsilon(\chi, \psi, t \cdot \dd x)
        = t \cdot \varepsilon(\chi, \psi, \dd x).
      \]
    \item Let $a \in F^\times$, and let $\psi_a$ denote the character defined
      by $\psi_a(x) = \psi(ax)$. Show that
      \[
        \varepsilon(\chi, \psi_a, \dd x)
        = \chi(a)\abs{a}^{-1} \varepsilon(\chi, \psi, \dd x).
      \]
    % Changed from the book: the book has \nu(\pi^{m+n}) and refers to
    % Eq. (7.6) in the hint.
    % - Exponent: replacing \psi by \psi_a with a = \pi^k turns the conductor
    %   P^m into P^{m-k}; comparing with (b) shows that the exponent must be
    %   n - m (Tate's a(\chi) + n(\psi) with n(\psi) = -m), not m + n.
    % - Hint: the book refers to Eq. (7.6), but the relevant formula is (7.5).
    \item Let $F$ be non-Archimedean with unique prime ideal $P$, and let
      $P^n$ and $P^m$ be the conductors of $\chi$ and $\psi$, respectively.
      Then show for every unramified character $\nu$ of $F^\times$ that
      \[
        \varepsilon(\chi\nu, \psi, \dd x)
        = \nu(\pi^{n-m}) \varepsilon(\chi, \psi, \dd x)
      \]
      where $\pi$ is a uniformizing parameter for $\gotho_F$. Note that the
      first factor on the right is well-defined because $\nu$ is unramified.
      [\emph{Hint}: Use the explicit formula given in \RV{Eq.~(7.5)}.]
      (\emph{Note}: This differs from the book, which has $\nu(\pi^{m+n})$,
      inconsistent with part~(b). Also, the book's hint refers to
      \RV{Eq.~(7.6)}, but the relevant formula is \RV{Eq.~(7.5)}.)
  \end{enumerate}
\end{exercise}

\begin{solution}
  Recall that
  \[\varepsilon(\chi, \psi, \dd x) =
    \frac{L(\chi)Z(\dual{f}, \chi^\vee )}{L (\chi ^ \vee) Z(f, \chi)}, \]
  where $\chi^\vee = \chi^{-1}|\cdot|$, $Z(f, \chi) = \int_{F^\times} f(x) \chi(x) \dd x$.

  (a) Let $\dual{f_t}$ be the fourier transform of $f$ with respect to the measure $t \dd x$.
  Then \[\dual{f_t} (x) = \int_{F} f(y) \psi(x y) t \dd x = t \dual{f} (x). \]

  It follows that $Z(\dual{f_t}, \chi^\vee) = t Z(\dual{f}, \chi^\vee)$. Since $L(\chi)$,
  $L(\chi^\vee)$ is independent on measure, then
  \[\varepsilon (\chi, \psi, t \dd x) =
  \frac{L(\chi)Z(\dual{f_t}, \chi^\vee )}{L (\chi ^ \vee) Z(f, \chi)} =
  \frac{L(\chi)t \cdot Z(\dual{f}, \chi^\vee)}{L(\chi^\vee) Z(f, \chi)} =
  t \cdot \varepsilon(\chi, \psi, \dd x).  \]

  (b) Let $\dual{f_a}$ be the fourier transform of $f$ with respect to the character $\psi_a$.
  Note that
  \[\dual{f_a} (x) = \int_{F} f(y) \psi_a(x y) \dd y = \int_{F} f(y) \psi(a x y) \dd y
  = \dual{f} (a x).  \]
  Then
  \[Z(\dual{f_a}, \chi^\vee) = \int_{F^\times} \dual{f_a} (x) \chi^\vee (x) \dd x =
  \chi^\vee(a) ^{-1} \int_{F^\times} \dual{f} (a x)\chi^\vee(a x) \dd ax =
  \chi^\vee (a)^{-1} Z(\dual{f}, \chi^\vee). \]
  It follows that
  \[\varepsilon (\chi, \psi_a, \dd x) = \chi^\vee(a)^{-1} \varepsilon(\chi, \psi, \dd x) =
  \chi(a) |a|^{-1} \varepsilon(\chi, \psi, \dd x).  \]

  (c)
\end{solution}

\begin{exercise}[9]
  \label{ex:ch7-9}
  Let $F$ be a local field with standard character $\psi$ and self-dual
  measure $\dd x$, and let $\chi = \omega\abs{\cdot}^s$ be a
  (quasi-)character on $F^\times$ with $\omega$ unitary. Put
  \[
    W(\omega) = \varepsilon(\chi, \psi, \dd x)\big|_{s=1/2}.
  \]
  \begin{enumerate}[label=(\alph*)]
    \item By using the functional equation, show that
      \[
        W(\omega) W(\overline{\omega}) = \omega(-1).
      \]
      Conclude that $\abs{W(\omega)} = 1$. [\emph{Hint}: First prove that
      $W(\overline{\omega}) = \omega(-1)\overline{W(\omega)}$.]
    \item Using \RV{Section 7.1}, show that for $F = \RR$, $W(\omega) = 1$
      (respectively, $-i$) for $\omega = 1$ (respectively, $\omega = \sgn$).
      Show further that when $F = \CC$, we have $W(\omega) = 1$.
    % The scan appears to read \psi(x\pi^{d-n}); since \psi has conductor
    % \calD_F^{-1} = \pi^{-d}\gotho_F, the exponent must be -d-n
    % (cf. Exercise 10(a), where c\gotho_F = \pi^n\calD_F).
    \item Let $F$ be non-Archimedean with uniformizing parameter $\pi$, and
      let $\omega$ as above have conductor $\pi^n\gotho_F$. Put
      \[
        G = \sum_{x \in U/U_n} \overline{\omega}(x) \psi(x\pi^{-d-n})
      \]
      where $\pi^d\gotho_F = \calD_F$ (the different),
      $U_r = 1 + \pi^r\gotho_F$ ($r > 0$), and $U$ is the full group of units
      in $\gotho_F$. Show directly that $\abs{G}^2 = q^n$. Then appeal to
      \RV{Eq.~(7.6)} to deduce (once again) that $\abs{W(\omega)} = 1$.
  \end{enumerate}
\end{exercise}

\begin{exercise}[10]
  \label{ex:ch7-10}
  (Tate) Let $F$ be a non-Archimedean local field with uniformizing parameter
  $\pi$, and let $\omega$ be a unitary character of $F^\times$ with conductor
  $\pi^n\gotho_F$. Let $\gotha$ be an ideal of $\gotho_F$ such that
  $\gotha^2$ divides $\pi^n\gotho_F$. Put
  $\gothb = \gotha^{-1}\pi^n\gotho_F \subseteq \gotho_F$.
  \begin{enumerate}[label=(\alph*)]
    \item Show that there exists an element $c \in F$ such that
      $c\gotho_F = \pi^n\calD_F$ and also
      \[
        \omega(1+t) = \psi(c^{-1}t)
      \]
      for all $t \in \gothb$. [\emph{Hint}: Suppose that
      $\gotha \neq \gotho_F$, so that $\pi$ divides $\gotha$. If
      $t, z \in \gothb$, then $tz \in \pi^n\gotho_F$. Consequently,
      $\omega(1+t)\omega(1+z) = \omega(1+t+z)$, and therefore the map that
      sends $t$ to $\omega(1+t)$ is a character of the additive group
      $\gothb$, and this extends to one on $F$. Now appeal to the isomorphism
      of $F$ with its dual.]
    \item For $c$ as in the previous part and $F$ now assumed to have
      characteristic zero, show that
      \[
        W(\omega) = N(\gothb\gotha^{-1})^{-1/2}
        \sum_{x \in (1+\gotha)/(1+\gothb)}
        \overline{\omega}(c^{-1}x)\psi(c^{-1}x).
      \]
      \textsc{Note}. When $\gotha = \gotho_F$, this result is identical to
      \RV{Eq.~(7.6)}.
    \item For $\omega$ unramified, show that $W(\omega) = \omega(\pi)^d$ if
      $\pi^d\gotho_F = \calD_F$.
    % The scan reads |W(\omega)|_p; the subscript should be the place v.
    \item Let $E$ be a number field containing $W(\omega)$. Then show that
      for every place $v$ of $E$ not dividing the residual characteristic $p$
      of $F$, we have $\abs{W(\omega)}_v = 1$. Using the Artin product
      formula, conclude that if $E$ has a unique place $u$ above $p$, then
      also $\abs{W(\omega)}_u = 1$.
    \item Suppose that $\pi^n\gotho_F = \pi\gotha$. Then, using part~(b),
      show that $z = (\omega(c^{-1})W(\omega))^2$ lies in the cyclotomic
      extension $E = \QQ(e^{2\pi i/p^r})$ for some $r \ge 1$. Show further,
      using part~(d), that $z$ must be a root of unity.
    \item (Lamprecht, Dwork) Suppose that $\omega$ is either
      \emph{unramified} (that is, $n = 0$) or \emph{wildly ramified} (that
      is, $n \ge 2$). Show that $W(\omega)$ is a root of unity.
      [\emph{Hint}: This is clear if $n$ is even. When $n$ is odd, write
      $\pi^n\gotho_F = \pi\gotha$ and apply the previous part.]
  \end{enumerate}
\end{exercise}
