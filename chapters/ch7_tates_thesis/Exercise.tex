\begin{exercise}[1]
  \label{ex:ch7-1}
  Let $F$ be a local field and let $\psi$ be a nontrivial unitary character on
  $F$; that is, a continuous, nontrivial homomorphism from $F$ as an additive
  group into $S^1$, the circle group. For each $a \in F$, define
  $\psi_a \colon F \to S^1$ by
  \[
    \psi_a(x) = \psi(ax).
  \]
  \begin{enumerate}[label=(\alph*)]
    \item Show that $\psi_a$ is again a unitary character of $F$, and that
      $\psi_a$ is trivial if and only if $a = 0$.
    % The book writes "\psi \mapsto \psi_a" (typo).
    \item Show that $a \mapsto \psi_a$ defines a continuous, injective
      homomorphism $\alpha_\psi$ of $F$ into its Pontryagin dual $\dual{F}$.
    % The book writes "for all b \in F" (typo).
    \item Show that the image of $\alpha_\psi$ is dense in $\dual{F}$.
      [\emph{Hint}: Show that $\psi_a(b) = 1$ for all $a \in F$ if and only if
      $b = 0$.]
    \item Show that $\alpha_\psi$ is bicontinuous.
    \item Show that $\alpha_\psi(F)$ is a complete, hence closed, subgroup of
      $\dual{F}$. Conclude that $\alpha_\psi$ is an isomorphism of topological
      groups.
  \end{enumerate}
\end{exercise}

\begin{solution}
(a) Multiplication by $a$, $x \mapsto ax$, is a continuous endomorphism of the
additive group $F$. Hence $\psi_a$, its composite with $\psi$, is a continuous
homomorphism $F \to S^1$, i.e.\ a unitary character of $F$.

If $a = 0$, then $\psi_a(x) = \psi(0) = 1$ for all $x \in F$, so $\psi_a$ is
trivial. Conversely, let $a \neq 0$. Since $\psi$ is nontrivial, there is some
$y \in F$ with $\psi(y) \neq 1$, and then $\psi_a(a^{-1}y) = \psi(y) \neq 1$.
So $\psi_a$ is nontrivial.

(b) Recall that $\dual{F}$ has the compact-open topology. A basis of neighborhoods of a
character $\psi_0$ is given by the sets
\[W(\psi_0, C, \epsilon) = \{\psi \in \dual{F}:  |\psi(x) - \psi_0(x)| < \epsilon, \forall x \in C \}. \]
where $C\subset F$ is a compact set and $\epsilon > 0$.

First, reduce to continuity at $0$. Note that $\psi_{a + b} (x)= \psi_a (x) \cdot \psi_b(x)$.
Then we have that
\[|\psi_a(x)- \psi_b(x)| = |\psi_b|\cdot |\psi_{a-b}(x) - 1 | = |\psi_{a-b}(x) - 1|. \]
So it suffices to show that for each compact $C$ and $\epsilon > 0$, there is a $\delta >0$, such
that
% \[|a| < \delta \Rightarrow |\psi_a(x) - 1| < \epsilon. \]
$\forall |a| < \delta$, $|\psi_a(x) - 1| < \epsilon$.

Given an $\epsilon > 0$. Since $\psi$ is continuous at $0$ and $\psi(0) = 1$, then there is an $r > 0$, such that
$\forall |x| < r$, $|\psi(x) - 1|< \epsilon$. Since $C$ is compact, then $M = \max_{x \in C} \{ |x|\} < \infty$.
Choose $\delta = r / M$, then $\forall |a| < \delta$ and $\forall x \in C$, $|a x| < r$.
It follows that
\[|\psi_a(x) - 1| = |\psi(ax) - 1| < \epsilon. \]

Injectivity is followed from (a).

(c)
Note that
\[\forall a \in F, \psi_a(b) = 1, \Leftrightarrow \forall a \in F, \psi(a b) = 1, \Leftrightarrow b = 0. \]

Recall that we have the fact of Pontryagin dual: let $G$ be a locally compact abelian group
and $H$ be a proper closed subgroup of $G$. Then there is a nontrivial character $\chi$ of $G$, such that
$\chi|_H = 1$.

Let $H$ be the closure of the image of $\alpha_\psi$, i.e. $H = \overline{\alpha_\psi(F)}$.
Assume that $H \neq \dual F$. Then we have a character $\chi \in \dual{\dual F}$ such that
$\chi$ is trivial on $H$. And by Pontryagin duality $F \equiv \dual{\dual{F}}$, every character
of $\dual F$ is evaluation at some point $b \in F$, i.e. $\psi \mapsto \psi (b)$.
Then we have that $\chi = (\psi \mapsto \psi(b_0))$ for some $b_0 \in F$. Since
$\chi$ is trivial on $H$ then $\forall a \in F, \psi_a(b_0) = 1$. It follows that
$b_0 = 0$, which contradicts to $\chi$ is a nontrivial character of $\dual{F}$.

(d) First, reduce to continuity at $0 \in F$. Note that $\psi_a \in \psi_{a_0} W(1, C, \epsilon)$ if and only if
$\psi_{a - a_0} \in W(1, C, \epsilon)$. Then it suffices to show that $\forall r > 0$, there are a
compact $C$ and $\epsilon > 0$, such that
\[\psi_a\in W(1, C, \epsilon) \Rightarrow |a|< r. \]

Since $\psi$ is nontrivial, pick $y \in F$ such that $\psi(y) \neq 1$, and set
$\epsilon = |\psi(y) - 1| > 0$. Now choose $C \coloneqq \{x \in F: |x| \le |y|/r \}$.
Since $C$ is closed, then $C$ is compact.

Now suppose that $\psi_a \in W(1, C, \epsilon)$ but $|a| \ge r$. Then $a \neq 0$ and
$x \coloneqq a^{-1} y$ satisfies $|x| = |y|/ |a| \le |y| / r$, so $x \in C$. But
\[|\psi_a(x) - 1| = |\psi(y) - 1| = \epsilon, \]
which contradicts to $\psi_a\in W(1, C, \epsilon)$. Therefore, $|a| < r$.

(e) Given a Cauchy sequence $\{\psi_{a_n}\}_n$ in $\alpha_\psi(F)$. Then we have that
for every neighborhood $W$ of $1$ there is an $N$ such that $\psi_{a_n - a_m} \in W$
for all $n, m \ge N$. By (d), there is a neighborhood $W$ such that
\[\psi_b \in W \Rightarrow |b| < r. \]

Then we have that $\{a_n\}_n$ is a Cauchy sequence in $F$. Since $F$ is complete,
then $\{a_n\}_n$ converges uniquely to $a \in F$. By (b), we have that $\psi_{a_n}$ converges
to $\psi_a$. Since $\dual{F}$ is Hausdorff, the limit is unique.
Therefore, $\alpha_{\psi} (F)$ is complete. And since every complete subspace of a Hausdorff space is
closed, then $\alpha_{\psi} (F)$ is closed. In conclusion, $\alpha_\psi$ is an isomorphism
of topological groups.
\end{solution}

\begin{exercise}[2]
  \label{ex:ch7-2}
  Let $F$ be a non-Archimedean local field with ring of integers $\gotho_F$
  and maximal ideal $P$. Fix a nontrivial unitary character $\psi$ of $F$.
  Let $P^m$ be the conductor of $\psi$; that is, the largest fractional ideal
  $P^m$ on which $\psi$ is trivial. Let $\alpha_\psi$, as defined in
  \cref{ex:ch7-1}, be the isomorphism between $F$ and its Pontryagin dual.
  \begin{enumerate}[label=(\alph*)]
    \item Show that via $\alpha_\psi$, $P^m$ identifies with the Pontryagin
      dual of $\dual{\gotho}_F$. (This is also the inverse different of $F$,
      which soon makes its appearance.)
    \item Let $n \ge 1$ and $x \in F$. Show that $xP^n \subseteq \ker(\psi)$
      if and only if $x \in P^{m-n}$.
  \end{enumerate}
\end{exercise}

\begin{exercise}[3]
  \label{ex:ch7-3}
  In this exercise we construct nontrivial characters for non-Archimedean
  local fields, including the standard character for $\QQ_p$ and its
  extensions, and the standard character for the completion of $\FF_p(t)$ and
  its extensions, at the place defined by $t^{-1}$.
  \begin{enumerate}[label=(\alph*)]
    \item Given any $x \in \QQ_p$, let $n$ be the smallest nonnegative integer
      such that $p^n x$ lies in $\ZZ_p$. Let $r \in \ZZ$ be such that
      $r \equiv p^n x \pmod{p^n}$. Put
      \[
        \psi(x) = e^{2\pi i r/p^n}.
      \]
      Show that $\psi \colon \QQ_p \to S^1$ is a nontrivial unitary character
      of conductor $\ZZ_p$ and that it identifies with the composite map
      $\QQ_p \to \QQ_p/\ZZ_p \to \QQ/\ZZ \to S^1$.
    % The scan appears to read a_1 (the coefficient of t) in the exponent
    % below. With a_1 the conductor is indeed F_p[[t^{-1}]], but then
    % Exercise 5(d) fails. The character compatible with Exercises 5 and 6
    % (the residue of x dt at infinity) is x |-> e^{-2\pi i a_{-1}/p}, whose
    % conductor is t^{-2} F_p[[t^{-1}]].
    \item Let $K_\infty$ denote the completion of $K = \FF_p(t)$ defined by
      the place $t^{-1}$. Show that $K_\infty = \FF_p((t^{-1}))$; that is,
      that each $x \in K_\infty$ can be uniquely represented by a formal power
      series
      \[
        \sum_{n=-\infty}^{r} a_n t^n
      \]
      with $a_n \in \ZZ/p\ZZ$ for all $n$ less than or equal to the integer
      $r$. Put
      \[
        \psi_\infty(x) = e^{2\pi i a_1/p} \in S^1.
      \]
      Show that $\psi_\infty$ is a nontrivial unitary character of $K_\infty$
      of conductor $\FF_p[[t^{-1}]]$.
    \item Let $\pi$ be any irreducible polynomial in $\FF_p[t] \subseteq K$,
      and let $K_\pi$ denote the completion of $K$ defined by the prime ideal
      $(\pi)$, with residue field $k$, which we may identify with a
      transversal of $(\pi)$ in $\FF_p[t]$. Show that every $x \in K_\pi$ can
      be written in the form
      \[
        \sum_{n=r}^{+\infty} a_n \pi^n
      \]
      for some integer $r$, with each coefficient $a_n \in k$. Next put
      \[
        \psi_\pi(x) = e^{2\pi i \tr_{k/\FF_p}(a_{-1})/p}.
      \]
      Show that $\psi_\pi$ is a nontrivial unitary character of $K_\pi$.
    \item Let $F$ be any non-Archimedean local field. Show that $F$ is a
      finite separable extension of a local field $F_0$ as in parts~(a), (b),
      or~(c) above. Let $\psi_0$ denote the corresponding character of $F_0$
      defined therein. Given $x \in F$, set
      \[
        \psi(x) = \psi_0(\tr_{F/F_0}(x)).
      \]
      Show that $\psi$ is a nontrivial unitary character of $F$.
    \item Continuing in the same context, show that when the characteristic
      of $F$ is zero, the conductor of $\psi$ is the inverse different
      $\calD_F^{-1}$. In this case show also that for $f$ the characteristic
      function of $\gotho_F$, the Fourier transform of $f$ is
      $N(\calD_F)^{-1/2}$ times the characteristic function of
      $\calD_F^{-1}$.
  \end{enumerate}
\end{exercise}

\begin{exercise}[4]
  \label{ex:ch7-4}
  We shall now construct the standard character on $\AAA_K/K$ for an
  algebraic number field $K$.
  \begin{enumerate}[label=(\alph*)]
    \item For each rational prime $p$, let $\psi_p$ denote the standard
      character of $\QQ_p$ defined above. Also define
      $\psi_\infty \colon \RR \to S^1$ by $\psi_\infty(x) = e^{-2\pi i x}$.
      Show that
      \[
        \psi \colon \AAA_\QQ \to S^1, \qquad
        (x_v) \mapsto \prod_v \psi_v(x_v)
      \]
      is a nontrivial unitary character on $\AAA_\QQ$, which is trivial on
      $\QQ$ (embedded diagonally). [\emph{Hint}: Use the Artin product
      formula.]
    \item For any number field $K$, define $\psi_K \colon \AAA_K \to S^1$ by
      \[
        \psi_K(x) = \prod_v \psi_v(x_v)
      \]
      where $v$ ranges over all the places of $K$ and $\psi_v$ is the standard
      character on the local field $K_v$, as constructed above. Show that
      $\psi_K$ is a nontrivial unitary character on $\AAA_K$.
    \item Show also that $\psi_K(x) = \psi(\tr(x))$ where $\tr$ denotes the
      adelic trace map from $K$ to $\QQ$. Conclude from part~(a) that
      $\psi_K$ is trivial on $K$.
  \end{enumerate}
\end{exercise}

\begin{exercise}[5]
  \label{ex:ch7-5}
  We shall now construct the standard character for the completion $K_v$ of
  $K = \FF_p(t)$ at any \emph{finite} place $v$.
  \begin{enumerate}[label=(\alph*)]
    \item Let $\pi(t)$ be an irreducible element of $\FF_p[t]$ of degree
      $d \ge 1$. Show that $K$ has a unique place $v$ such that the polynomial
      $\pi(t)$ generates the maximal ideal of
      $\gotho_v = \{\alpha \in K_v : \abs{\alpha}_v \le 1\}$. Show further
      that $q_v = \operatorname{Card}(\gotho_v/\pi\gotho_v) = p^d$, and that
      the polynomials in $\FF_p[t]$ of degree less than $d$ in fact generate
      $\gotho_v/\pi\gotho_v$.
    % Note: as defined, K^{(v)} contains the constants, so
    % K^{(v)} \cap \gotho_v = F_p and the sum below is not direct; part (c),
    % where deg z < nd, implicitly uses the strict inequality |y|_\infty < 1.
    \item Given any place $v$ of $K$ (possibly infinite), let us define
      \[
        K^{(v)} = \{x \in K : \abs{x}_u \le 1 \ \forall u \neq v\}.
      \]
      Show that $K_v = K^{(v)} \oplus \gotho_v$. [\emph{Hint}: By replacing
      $t$ by $t^{-1}$ if necessary, one may assume that $v$ is defined by an
      irreducible polynomial $\pi(t) \in \FF_p[t]$ and then apply the previous
      part.]
    \item Let $v$ be any finite place of $K$; that is, one defined by an
      irreducible polynomial $\pi(t) \in \FF_p[t]$ of positive degree $d$.
      Show that every element $y$ of $K^{(v)}$ can be written as $z/\pi^n$
      for some $n \ge 0$ and polynomial $z = z(t) \in \FF_p[t]$ of the form
      \[
        c_1 t^{nd-1} + c_2 t^{nd-2} + \cdots + c_{nd}
      \]
      with $c_j \in \FF_p$ for all $j$, possibly all zero. If
      $x = y + \alpha \in K_v$ with $\alpha \in \gotho_v$, put
      \[
        \psi_v(x) = e^{2\pi i c_1/p}.
      \]
      (This is of course well-defined despite the status of $c_1$ as a residue
      class modulo the prime $p$.) Show that $\psi_v$ is a nontrivial
      character of $K_v$ of conductor $\gotho_v$.
    % See the note at Exercise 3(b): the identity \psi_v(y)\psi_\infty(y) = 1
    % requires \psi_\infty(x) = e^{-2\pi i a_{-1}/p}. The book writes f for z
    % in the hint.
    \item Let $v$, $\pi$, and $y$ be as in the previous part. Show that
      $y \equiv c_1 t^{-1} \pmod{t^{-2}}$. Conclude that
      $\psi_v(y)\psi_\infty(y) = 1$, where $\psi_\infty$ is as in part~(b) of
      \cref{ex:ch7-3}. [\emph{Hint}: Write $\pi$ as $t^d g$, with $g$ a
      polynomial in $t^{-1}$ of constant term $1$, so that
      $y = z g^{-n} t^{-nd}$.]
  \end{enumerate}
\end{exercise}
