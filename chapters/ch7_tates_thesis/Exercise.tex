\begin{exercise}[1]
  \label{ex:ch7-1}
  Let $K$ be a local field and let $\psi$ be a nontrivial unitary character on
  $K$; that is, a continuous, nontrivial homomorphism from $K$ as an additive
  group into $S^1$, the circle group. For each $a \in K$, define
  $\psi_a \colon K \to S^1$ by
  \[
    \psi_a(x) = \psi(ax).
  \]
  \begin{enumerate}[label=(\alph*)]
    \item Show that $\psi_a$ is again a unitary character of $K$, and that
      $\psi_a$ is trivial if and only if $a = 0$.
    % The book writes "\psi \mapsto \psi_a" (typo).
    \item Show that $a \mapsto \psi_a$ defines a continuous, injective
      homomorphism $\alpha_\psi$ of $K$ into its Pontryagin dual $\dual{K}$.
    % The book writes "for all b \in F" (typo).
    \item Show that the image of $\alpha_\psi$ is dense in $\dual{K}$.
      [\emph{Hint}: Show that $\psi_a(b) = 1$ for all $a \in K$ if and only if
      $b = 0$.]
    \item Show that $\alpha_\psi$ is bicontinuous.
    \item Show that $\alpha_\psi(K)$ is a complete, hence closed, subgroup of
      $\dual{K}$. Conclude that $\alpha_\psi$ is an isomorphism of topological
      groups.
  \end{enumerate}
\end{exercise}

\begin{solution}
(a) Multiplication by $a$, $x \mapsto ax$, is a continuous endomorphism of the
additive group $K$. Hence $\psi_a$, its composite with $\psi$, is a continuous
homomorphism $K \to S^1$, i.e.\ a unitary character of $K$.

If $a = 0$, then $\psi_a(x) = \psi(0) = 1$ for all $x \in K$, so $\psi_a$ is
trivial. Conversely, let $a \neq 0$. Since $\psi$ is nontrivial, there is some
$y \in K$ with $\psi(y) \neq 1$, and then $\psi_a(a^{-1}y) = \psi(y) \neq 1$.
So $\psi_a$ is nontrivial.

(b) Recall that $\dual{K}$ has the compact-open topology. A basis of neighborhoods of a
character $\psi_0$ is given by the sets
\[W(\psi_0, C, \epsilon) = \{\psi \in \dual{K}:  |\psi(x) - \psi_0(x)| < \epsilon, \forall x \in C \}. \]
where $C\subset K$ is a compact set and $\epsilon > 0$.

First, reduce to continuity at $0$. Note that $\psi_{a + b} (x)= \psi_a (x) \cdot \psi_b(x)$.
Then we have that
\[|\psi_a(x)- \psi_b(x)| = |\psi_b|\cdot |\psi_{a-b}(x) - 1 | = |\psi_{a-b}(x) - 1|. \]
So it suffices to show that for each compact $C$ and $\epsilon > 0$, there is a $\delta >0$, such
that
% \[|a| < \delta \Rightarrow |\psi_a(x) - 1| < \epsilon. \]
$\forall |a| < \delta$, $|\psi_a(x) - 1| < \epsilon$.

Given an $\epsilon > 0$. Since $\psi$ is continuous at $0$ and $\psi(0) = 1$, then there is an $r > 0$, such that
$\forall |x| < r$, $|\psi(x) - 1|< \epsilon$. Since $C$ is compact, then $M = \max_{x \in C} \{ |x|\} < \infty$.
Choose $\delta = r / M$, then $\forall |a| < \delta$ and $\forall x \in C$, $|a x| < r$.
It follows that
\[|\psi_a(x) - 1| = |\psi(ax) - 1| < \epsilon. \]

Injectivity is followed from (a).

(c)
Note that
\[\forall a \in K, \psi_a(b) = 1, \Leftrightarrow \forall a \in K, \psi(a b) = 1, \Leftrightarrow b = 0. \]

Recall that we have the fact of Pontryagin dual: let $G$ be a locally compact abelian group
and $H$ be a proper closed subgroup of $G$. Then there is a nontrivial character $\chi$ of $G$, such that
$\chi|_H = 1$.

Let $H$ be the closure of the image of $\alpha_\psi$, i.e. $H = \overline{\alpha_\psi(K)}$.
Assume that $H \neq \dual K$. Then we have a character $\chi \in \dual{\dual K}$ such that
$\chi$ is trivial on $H$. And by Pontryagin duality $K \equiv \dual{\dual{K}}$, every character
of $\dual K$ is evaluation at some point $b \in K$, i.e. $\psi \mapsto \psi (b)$.
Then we have that $\chi = (\psi \mapsto \psi(b_0))$ for some $b_0 \in K$. Since
$\chi$ is trivial on $H$ then $\forall a \in K, \psi_a(b_0) = 1$. It follows that
$b_0 = 0$, which contradicts to $\chi$ is a nontrivial character of $\dual{K}$.



\end{solution}

\begin{exercise}[2]
  \label{ex:ch7-2}
  Let $K$ be a non-Archimedean local field with ring of integers $\gotho_K$
  and maximal ideal $P$. Fix a nontrivial unitary character $\psi$ of $K$.
  Let $P^m$ be the conductor of $\psi$; that is, the largest fractional ideal
  $P^m$ on which $\psi$ is trivial. Let $\alpha_\psi$, as defined in
  \cref{ex:ch7-1}, be the isomorphism between $K$ and its Pontryagin dual.
  \begin{enumerate}[label=(\alph*)]
    \item Show that via $\alpha_\psi$, $P^m$ identifies with the Pontryagin
      dual of $\dual{\gotho}_K$. (This is also the inverse different of $K$,
      which soon makes its appearance.)
    \item Let $n \ge 1$ and $x \in K$. Show that $xP^n \subseteq \ker(\psi)$
      if and only if $x \in P^{m-n}$.
  \end{enumerate}
\end{exercise}
