% Exercises are numbered <chapter>.<n>, matching the book.
% Usage:
%   \begin{exercise}      -> next number in this chapter
%   \begin{exercise}[7]   -> jump to Exercise <chapter>.7 (for skipped ones)
\theoremstyle{definition}
\newtheorem{exerciseinner}{Exercise}[chapter]
\newenvironment{exercise}[1][]{%
  \if\relax\detokenize{#1}\relax\else
    \setcounter{exerciseinner}{\numexpr#1-1\relax}%
  \fi
  \begin{exerciseinner}}{\end{exerciseinner}}
\crefname{exerciseinner}{Exercise}{Exercises}

\newenvironment{solution}{\begin{proof}[Solution]}{\end{proof}}
